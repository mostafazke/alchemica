# Plan 11-03 — AdMob Service Module

## Goal
Create `src/lib/effects/admob.ts` encapsulating the full AdMob lifecycle:
initialize, preload, show, reward, re-preload. Web-guarded. Exports
`isAdReady` Svelte store and `initAdMob()` / `requestAdHint()` functions.

## Tasks

### Task 1: Create `src/lib/effects/admob.ts`

```ts
import { writable } from 'svelte/store';
import { Capacitor } from '@capacitor/core';
import { AdMob, RewardAdPluginEvents } from '@capacitor-community/admob';
import { hintBalance } from '../stores/game.js';

const REWARD_AD_ID = 'ca-app-pub-3940256099942544/5224354917'; // test ID

/** True when a rewarded ad is loaded and ready to show. */
export const isAdReady = writable<boolean>(false);

let listenersRegistered = false;

function registerListeners(): void {
  if (listenersRegistered) return;
  listenersRegistered = true;

  AdMob.addListener(RewardAdPluginEvents.Loaded, () => {
    isAdReady.set(true);
  });

  AdMob.addListener(RewardAdPluginEvents.FailedToLoad, () => {
    isAdReady.set(false);
  });

  AdMob.addListener(RewardAdPluginEvents.Rewarded, () => {
    hintBalance.update((n) => n + 1);
    // Preload next ad immediately after reward
    AdMob.prepareRewardVideoAd({ adId: REWARD_AD_ID }).catch(() => {
      isAdReady.set(false);
    });
  });

  AdMob.addListener(RewardAdPluginEvents.FailedToShow, () => {
    isAdReady.set(false);
    // Try to reload
    AdMob.prepareRewardVideoAd({ adId: REWARD_AD_ID }).catch(() => { /* ignore */ });
  });
}

/** Initialize AdMob and preload first rewarded ad. Native-only — no-op on web. */
export async function initAdMob(): Promise<void> {
  if (!Capacitor.isNativePlatform()) return;

  try {
    await AdMob.initialize();
    registerListeners();
    await AdMob.prepareRewardVideoAd({ adId: REWARD_AD_ID });
  } catch {
    // AdMob init failure is non-fatal — ad button shows 'No ad available'
    isAdReady.set(false);
  }
}

/** Show the rewarded ad. No-op on web or when ad is not ready. */
export async function requestAdHint(): Promise<void> {
  if (!Capacitor.isNativePlatform()) return;
  isAdReady.set(false); // immediately disable button while ad plays
  try {
    await AdMob.showRewardVideoAd();
  } catch {
    // Show failed — reset state; listener FailedToShow handles reload
    isAdReady.set(false);
  }
}
```

### Task 2: Call `initAdMob()` in `src/routes/+layout.svelte`

In the `onMount` callback (alongside the screen.orientation lock):
```ts
import { initAdMob } from '$lib/effects/admob.js';

onMount(() => {
  // ... existing orientation lock ...
  initAdMob(); // non-blocking; no-op on web
});
```
