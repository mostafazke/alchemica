      await purchaseRemoveAds();

        setTimeout(() => { purchaseStatus = null; }, 4000);
      }
    }

    async function handleRestore() {
      await restorePurchases();
      const err = $purchaseError;
      if (!err && !$isPurchasing) {
        purchaseStatus = { ok: true, message: 'Purchases restored.' };
        setTimeout(() => { purchaseStatus = null; }, 4000);
      }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User tap on purchase button → iap.ts | UI input triggers native billing sheet |
| $purchaseError store → displayed text | Error string from RC SDK rendered in DOM |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-12-01 | Injection | $purchaseError → DOM | mitigate | Svelte renders store values as text content (not innerHTML). Even if an error string contained HTML, Svelte's template escaping prevents XSS. |
| T-12-02 | Tampering | isNative guard on Purchases section | mitigate | `Capacitor.isNativePlatform()` is evaluated at runtime. On web builds, the entire section is never rendered — no purchase buttons are accessible on PWA. |
| T-12-03 | Denial of Service | $isPurchasing disables buttons | mitigate | isPurchasing is set to false in iap.ts `finally` blocks — buttons re-enable even after errors. No permanent disabled state. |
| T-12-04 | Spoofing | "Restore Purchases" button | accept | Restore calls RC server-side validation — RC re-verifies purchase history against Google Play receipts. Client cannot fake a restore. |
</threat_model>
