    Create src/lib/effects/particles.ts — a typed, pooled particle system based on alchemica.html.

    Key improvements over the original:
    1. Object pooling: pre-allocate MAX_PARTICLES and reuse them (set `active = false` instead of splicing)
    2. Scale particle count with `Math.max(1, navigator.hardwareConcurrency / 2)` multiplier
    3. Pause on `document.visibilitychange` → `document.hidden`
    4. Accept canvas element as parameter (not hard-coded getElementById)

    ```typescript
    interface Particle {
      x: number; y: number;
      vx: number; vy: number;
      life: number; maxLife: number;
      decay: number; size: number;
      color: string;
      type: 'burst' | 'bubble' | 'fail';
      active: boolean;
    }

    const MAX_PARTICLES = 120;
    const pool: Particle[] = Array.from({ length: MAX_PARTICLES }, () => ({
      x: 0, y: 0, vx: 0, vy: 0, life: 0, maxLife: 1,
      decay: 0.02, size: 3, color: '#4af0c0', type: 'burst', active: false
    }));

    let canvas: HTMLCanvasElement | null = null;
    let ctx: CanvasRenderingContext2D | null = null;
    let rafId = 0;
    let paused = false;

    export function initParticles(canvasEl: HTMLCanvasElement) {
      canvas = canvasEl;
      ctx = canvas.getContext('2d');
      document.addEventListener('visibilitychange', () => {
        paused = document.hidden;
        if (!paused && pool.some(p => p.active)) animate();
      });
    }

    function acquireParticle(): Particle | null {
      return pool.find(p => !p.active) ?? null;
    }

    const BURST_COLORS = ['#4af0c0','#ffe44a','#ff6b6b','#5ab4ff','#d05aff','#96c84a'];
    const perfMultiplier = Math.max(0.5, Math.min(2, (navigator.hardwareConcurrency ?? 4) / 4));

    export function triggerSuccessParticles(cx: number, cy: number) {
      const burstCount = Math.round(60 * perfMultiplier);
      const bubbleCount = Math.round(20 * perfMultiplier);

      for (let i = 0; i < burstCount; i++) {
        const p = acquireParticle();
        if (!p) break;
        const angle = Math.random() * Math.PI * 2;
        const speed = 1.5 + Math.random() * 4;
        Object.assign(p, {
          x: cx, y: cy, vx: Math.cos(angle) * speed,
          vy: Math.sin(angle) * speed - 2, life: 1, maxLife: 1,
          decay: 0.015 + Math.random() * 0.02, size: 2 + Math.random() * 5,
          color: BURST_COLORS[Math.floor(Math.random() * BURST_COLORS.length)],
          type: 'burst', active: true
        });
      }
      for (let i = 0; i < bubbleCount; i++) {
        const p = acquireParticle();
        if (!p) break;
        Object.assign(p, {
          x: cx + (Math.random() - 0.5) * 80, y: cy + 30,
          vx: (Math.random() - 0.5), vy: -(1 + Math.random() * 2),
          life: 1, maxLife: 1, decay: 0.008 + Math.random() * 0.01,
          size: 3 + Math.random() * 6, color: '#4af0c040',
          type: 'bubble', active: true
        });
      }
      scheduleAnimate();
    }

    export function triggerFailParticles(cx: number, cy: number) {
      const count = Math.round(20 * perfMultiplier);
      for (let i = 0; i < count; i++) {
        const p = acquireParticle();
        if (!p) break;
        Object.assign(p, {
          x: cx + (Math.random() - 0.5) * 40, y: cy,
          vx: (Math.random() - 0.5) * 2, vy: (Math.random() - 0.5) * 2,
          life: 1, maxLife: 1, decay: 0.03, size: 2 + Math.random() * 3,
          color: '#ff405070', type: 'fail', active: true
        });
      }
      scheduleAnimate();
    }

    function scheduleAnimate() {
      cancelAnimationFrame(rafId);
      if (!paused) rafId = requestAnimationFrame(animate);
    }

    function animate() {
      if (!canvas || !ctx || paused) return;

      // Resize canvas to parent
      const parent = canvas.parentElement;
      if (parent) {
        canvas.width = parent.offsetWidth;
        canvas.height = parent.offsetHeight;
      }

      ctx.clearRect(0, 0, canvas.width, canvas.height);
      let anyActive = false;

      for (const p of pool) {
        if (!p.active) continue;
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.08; // gravity
        p.vx *= 0.98;
        p.life -= p.decay;

        if (p.life <= 0) { p.active = false; continue; }
        anyActive = true;

        ctx.save();
        ctx.globalAlpha = Math.max(0, p.life);
        if (p.type === 'bubble') {
          ctx.strokeStyle = '#4af0c0';
          ctx.lineWidth = 1;
          ctx.beginPath();
          ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
          ctx.stroke();
        } else {
          ctx.fillStyle = p.color;
          ctx.beginPath();
          ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
          ctx.fill();
        }
        ctx.restore();
      }

      if (anyActive) {
        rafId = requestAnimationFrame(animate);
      } else {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
      }
    }
    ```
