    Create src/lib/game/reactions.ts with a `resolveReaction` function:

    ```typescript
    import { REACTIONS, MULTI_REACTIONS } from '../data/reactions.js';
    import { ELEMENTS } from '../data/elements.js';
    import { get } from 'svelte/store';
    import { unlockedElements, discoveries, combo, score, lastSuccess } from '../stores/game.js';
    import type { Discovery } from '../types.js';

    export function resolveReaction(a: string, b: string): { result: string | null; isNew: boolean } {
      const key = `${a}+${b}`;
      const result = REACTIONS[key] ?? null;
      return { result, isNew: result !== null && !get(unlockedElements).has(result) };
    }

    export function applyReaction(a: string, b: string): string | null {
      const { result, isNew } = resolveReaction(a, b);

      if (result) {
        const currentCombo = get(combo);
        const currentLastSuccess = get(lastSuccess);

        // Update combo
        const newCombo = currentLastSuccess ? Math.min(currentCombo + 1, 8) : 1;
        combo.set(newCombo);
        lastSuccess.set(true);

        // Update score
        const points = isNew ? 100 * newCombo : 10 * newCombo;
        score.update(s => s + points);

        // Unlock and log discovery
        if (isNew) {
          unlockedElements.update(s => { s.add(result); return s; });
          const discovery: Discovery = {
            key: result,
            recipe: ELEMENTS[result]?.recipe ?? `${ELEMENTS[a]?.name} + ${ELEMENTS[b]?.name}`,
            timestamp: Date.now()
          };
          discoveries.update(d => [discovery, ...d]);
        } else {
          unlockedElements.update(s => { s.add(result); return s; });
        }
      } else {
        lastSuccess.set(false);
        combo.set(1);
      }

      return result;
    }
    ```
